ls -lah --full-time /srv/  # 人类可读大小 + 秒级时间（默认是 mtime）

# 在服务器加载镜像
docker load -i /srv/myapp.tar

# QQ 邮箱找回密码：首次在服务器执行一次，后续更新镜像会继续使用此配置文件
# 新依赖 dkimpy 会在 docker build 时随 requirements.txt 自动安装
# 授权码在提示时输入，不会显示，也不要填写到本文件或提交到 Git
mkdir -p /srv/myapp
if [ ! -f /srv/myapp/password-reset.env ]; then
  (
    umask 077
    read -r -s -p "请输入 QQ 邮箱 IMAP 授权码：" reset_imap_password
    printf '\n'
    if [ -z "$reset_imap_password" ]; then
      printf '授权码不能为空，请重新执行配置步骤。\n' >&2
      exit 1
    fi
    printf 'PASSWORD_RESET_ENABLED=true\nPASSWORD_RESET_IMAP_PASSWORD=%s\n' "$reset_imap_password" > /srv/myapp/password-reset.env
  )
fi
chmod 600 /srv/myapp/password-reset.env
# 确认以上步骤成功后再删除旧容器；不要把此配置文件放进 data 目录，避免替换 data 时丢失
# 更换授权码时，编辑 /srv/myapp/password-reset.env，再重新创建容器

# 先删旧容器
docker rm -f myapp

# 运行容器（把宿主机 /srv/myapp/data 挂到容器 /app/data）
docker run -d --name myapp -p 80:5000 \
  --env-file /srv/myapp/password-reset.env \
  -v /srv/myapp/data:/app/data \
  -v /srv/myapp/logs:/app/logs \
  myapp:latest
# 用ngix（推荐：添加了 statics volume 挂载）
docker run -d --name myapp \
  --network host \
  --env-file /srv/myapp/password-reset.env \
  -e FORWARDED_ALLOW_IPS="*" \
  -v /srv/myapp/data:/app/data \
  -v /srv/myapp/logs:/app/logs \
  -v /srv/myapp/statics:/app/app/statics \
  --memory="1.5g" \
  --memory-swap="2.5g" \
  --memory-reservation="1g" \
  --cpus="1.5" \
  --oom-kill-disable=false \
  myapp:latest
